<!-- From zoho-middleware/lib/recipe-scaling.js: -->
classifyUnit(raw) → { family: 'mass'|'volume'|'count'|null, norm }
  MASS_FACTORS = { kg: 1, g: 0.001 }   VOLUME_FACTORS = { l: 1, ml: 0.001 }
  COUNT_UNITS = ['pcs','ea','each','unit','pkg','pack']
ingredientLineCost(item, line) → { ok:true, convertedQty, cost } | { ok:false, error }
  mass/volume: convertedQty = qty * (factors[lineUnit] / factors[itemUnit]); 4dp round.
  cost = round(convertedQty * item.rate, 4dp). count: convertedQty = qty (pass-through).
  cross-family / unrecognised on either side → { ok:false } (fail closed).

<!-- CLIENT patterns to match (vanilla ES5, module scope, before the IIFE at brewpad.js:757). -->

     exported for tests in the module.exports block (js/brewpad.js:9183-9223). -->

<!-- CRITICAL SUBTLETY — the review's suggested `bpIngredientLineCost(ing, ing)` is INSUFFICIENT:
     the ingredient object carries ONE `unit` (the recipe-LINE unit) plus catalog-derived
     `rate`/`purchase_rate` (priced per the CATALOG item unit). enrichIngredientsWithCatalogRates
     (js/brewpad.js:233-247) only backfills `ing.unit` from the catalog when the line unit is BLANK
     (line 243) — it never records the catalog unit separately. Passing `ing` as BOTH item and line
     makes itemUnit === lineUnit → convertedQty === qty → NO conversion (silent no-op). The fix MUST

     as line so a g-line against a kg item actually converts. -->
