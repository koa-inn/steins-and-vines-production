**A. Check what is exportable from brewpad.js**

The STATUS_LABELS, STATUS_COLORS, and filterOpts variables are inside the IIFE scope of brewpad.js and may not be directly testable via require(). The existing test pattern (brewpad-auth.test.js) tests only the pure helper functions (`isSessionStale`, `isSessionExpired`) that are explicitly exported via `module.exports`.

For this phase, we need to test:
1. That STATUS_LABELS contains the `pending` key
2. That STATUS_COLORS maps `pending` to `neutral`
3. That the kiosk badge visibility condition is correct (source=kiosk AND status=pending)

Since STATUS_LABELS and STATUS_COLORS are inside the IIFE, they cannot be directly exported without restructuring. Instead, create a pure helper function that encapsulates the badge visibility logic and export it:

**Add to js/brewpad.js** (before the module.exports block at the bottom):

```javascript
/**
 * Determine if a batch should show the kiosk source badge.
 * Per D-11: visible only when source=kiosk AND status=pending.
 * @param {string} source - batch source value
 * @param {string} status - batch status value
 * @returns {boolean}
 */
function shouldShowKioskBadge(source, status) {
  return source === 'kiosk' && (status || '').toLowerCase() === 'pending';
}
```

Update the inline conditionals in the table and card render loops to use this function:
```javascript
// Replace: if (b.source === 'kiosk' && statusKey === 'pending')
// With:    if (shouldShowKioskBadge(b.source, statusKey))
```

Then add `shouldShowKioskBadge` to the `module.exports` block:
```javascript
if (typeof module !== 'undefined' && module.exports) {
  module.exports = {
    isSessionStale: isSessionStale,
    isSessionExpired: isSessionExpired,
    shouldShowKioskBadge: shouldShowKioskBadge
  };
}
```

**B. Create tests/frontend/brewpad-pending.test.js**

```javascript
'use strict';

var brewpad = require('../../js/brewpad');

describe('BrewPad pending batch helpers', function () {

  describe('shouldShowKioskBadge', function () {

    it('returns true when source is kiosk and status is pending', function () {
      expect(brewpad.shouldShowKioskBadge('kiosk', 'pending')).toBe(true);
    });

    it('returns false when source is manual', function () {
      expect(brewpad.shouldShowKioskBadge('manual', 'pending')).toBe(false);
    });

    it('returns false when status is not pending (active kiosk batch)', function () {
      expect(brewpad.shouldShowKioskBadge('kiosk', 'primary')).toBe(false);
      expect(brewpad.shouldShowKioskBadge('kiosk', 'secondary')).toBe(false);
      expect(brewpad.shouldShowKioskBadge('kiosk', 'complete')).toBe(false);
    });

    it('returns false when source is empty or undefined', function () {
      expect(brewpad.shouldShowKioskBadge('', 'pending')).toBe(false);
      expect(brewpad.shouldShowKioskBadge(undefined, 'pending')).toBe(false);
      expect(brewpad.shouldShowKioskBadge(null, 'pending')).toBe(false);
    });

    it('handles case-insensitive status', function () {
      expect(brewpad.shouldShowKioskBadge('kiosk', 'Pending')).toBe(true);
      expect(brewpad.shouldShowKioskBadge('kiosk', 'PENDING')).toBe(true);
    });

    it('returns false when both source and status are empty', function () {
      expect(brewpad.shouldShowKioskBadge('', '')).toBe(false);
    });

  });

});
```
