recipe-scaling.js exports (all pure, no I/O, no requires):

scaleIngredient(ing, factor) -> shallow clone of ing with quantity scaled
  - ing: { quantity, unit, item_id, item_name, ... }
  - continuous unit -> Math.round(rawQty * 10000) / 10000  (linear, float-drift-safe)
  - discrete unit   -> Math.max(1, Math.ceil(rawQty))
  - blank/unknown unit -> linear (D-03)

scaleIngredients(ingredients, factor) -> ingredients.map(scaleIngredient)

computeScaledRecipeTotal(recipe, scaledIngredients, catalogMap, saleType) -> number (2dp)
  - locked mode (pricing_mode==='locked' && locked_price>0):
      total = locked_price * recipe._scale_factor   (caller sets recipe._scale_factor)
  - dynamic mode: total = sum(scaledIng.quantity * catalogMap[item_id].rate)
  - if saleType === 'in-store': total += service_fee + materials_fee  (fixed, never scaled)
  - returns Math.round(total * 100) / 100

checkScaledStock(scaledIngredients, catalogMap) -> { ok: boolean, conflicts: [{ item_id, item_name, needed, stock, unit }] }
  - unknown item (not in catalogMap) -> skipped (not a conflict)
  - conflict when needed > stock_on_hand

CONTINUOUS_UNITS = ['kg', 'g', 'l', 'ml']
DISCRETE_UNITS   = ['pcs', 'each', 'unit', 'pkg', 'ft']
